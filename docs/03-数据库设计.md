# 新能源汽车测试平台 - 数据库设计

## 1. 数据库选型

| 数据类型 | 存储方案 | 说明 |
|----------|----------|------|
| 业务数据 | PostgreSQL | 关系型数据，事务支持 |
| 缓存数据 | Redis | 高性能缓存，任务队列 |
| 时序数据 | InfluxDB | 性能监控数据 |
| 日志数据 | Elasticsearch | 日志存储和检索 |
| 文件存储 | MinIO | 测试报告、日志文件 |

## 3. 详细表结构设计

### 3.1 用户与权限

#### users (用户表)
```sql
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255),
    role_id UUID REFERENCES roles(id),
    department VARCHAR(100),
    display_name VARCHAR(100),
    avatar_url VARCHAR(500),
    phone VARCHAR(20),
    ldap_dn VARCHAR(255),                    -- LDAP DN (如果使用LDAP认证)
    auth_type VARCHAR(20) DEFAULT 'local',   -- local, ldap, oauth2
    status VARCHAR(20) DEFAULT 'active',     -- active, inactive, locked
    last_login_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_users_username ON users(username);
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_status ON users(status);
```

#### roles (角色表)
```sql
CREATE TABLE roles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(50) NOT NULL UNIQUE,
    display_name VARCHAR(100),
    description TEXT,
    permissions JSONB NOT NULL DEFAULT '[]',  -- 权限列表
    is_system BOOLEAN DEFAULT FALSE,          -- 是否系统内置角色
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 预置角色
INSERT INTO roles (name, display_name, permissions, is_system) VALUES

('viewer', '访客', '["task:read", "env:read", "report:read"]', TRUE);
```

#### operation_logs (操作日志表)
```sql
CREATE TABLE operation_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    username VARCHAR(50),
    action VARCHAR(50) NOT NULL,              -- create, update, delete, execute
    resource_type VARCHAR(50) NOT NULL,       -- task, environment, report
    resource_id UUID,
    resource_name VARCHAR(200),
    details JSONB,                            -- 详细信息
    ip_address INET,
    user_agent TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_operation_logs_user_id ON operation_logs(user_id);
CREATE INDEX idx_operation_logs_resource ON operation_logs(resource_type, resource_id);
CREATE INDEX idx_operation_logs_created_at ON operation_logs(created_at DESC);
```

### 3.2 项目管理

#### projects (项目表)
```sql
CREATE TABLE projects (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    code VARCHAR(50) NOT NULL UNIQUE,         -- 项目编码
    description TEXT,
    owner_id UUID REFERENCES users(id),
    team_members JSONB DEFAULT '[]',          -- 团队成员ID列表
    settings JSONB DEFAULT '{}',              -- 项目配置
    status VARCHAR(20) DEFAULT 'active',      -- active, archived
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_projects_code ON projects(code);
CREATE INDEX idx_projects_status ON projects(status);
```

### 3.3 测试环境

#### test_hosts (测试主机表)
```sql
CREATE TABLE test_hosts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    description TEXT,
    ip_address INET NOT NULL,
    ssh_port INTEGER DEFAULT 22,
    username VARCHAR(50),
    auth_type VARCHAR(20) DEFAULT 'password', -- password, key
    password_encrypted VARCHAR(500),          -- 加密存储
    ssh_key_encrypted TEXT,                   -- SSH私钥(加密)
    work_directory VARCHAR(500) DEFAULT '/opt/test', -- 工作目录
    
    status VARCHAR(20) DEFAULT 'offline',     -- online, offline, busy, maintenance
    last_heartbeat TIMESTAMP WITH TIME ZONE,
    
    system_info JSONB,                        -- 系统信息 (OS, CPU, Memory等)
    tags JSONB DEFAULT '[]',                  -- 标签
    capabilities JSONB DEFAULT '[]',          -- 能力标签
    
    max_concurrent_tasks INTEGER DEFAULT 1,   -- 最大并发任务数
    current_task_count INTEGER DEFAULT 0,     -- 当前任务数
    
    project_id UUID REFERENCES projects(id),  -- 所属项目
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_test_hosts_status ON test_hosts(status);
CREATE INDEX idx_test_hosts_project_id ON test_hosts(project_id);
```

#### test_devices (测试设备表)
```sql
CREATE TABLE test_devices (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    description TEXT,
    device_type VARCHAR(50) NOT NULL,         -- ECU, VCU, BMS, MCU, Tbox, IVI
    model VARCHAR(100),                       -- 设备型号
    serial_number VARCHAR(100),               -- 序列号
    
    host_id UUID REFERENCES test_hosts(id),   -- 关联的测试主机
    connection_type VARCHAR(50),              -- ssh, serial, can, doip, uds
    connection_config JSONB,                  -- 连接配置
    
    current_version VARCHAR(50),              -- 当前软件版本
    hardware_version VARCHAR(50),             -- 硬件版本
    
    status VARCHAR(20) DEFAULT 'offline',     -- online, offline, busy, upgrading, error
    last_heartbeat TIMESTAMP WITH TIME ZONE,
    
    capabilities JSONB DEFAULT '[]',          -- 能力标签
    tags JSONB DEFAULT '[]',                  -- 标签
    
    project_id UUID REFERENCES projects(id),
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_test_devices_host_id ON test_devices(host_id);
CREATE INDEX idx_test_devices_status ON test_devices(status);
CREATE INDEX idx_test_devices_type ON test_devices(device_type);
```

#### environments (测试环境表 - 主机+设备组合)
```sql
CREATE TABLE environments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    description TEXT,
    env_type VARCHAR(50),                     -- HIL, SIL, 实车
    
    host_id UUID REFERENCES test_hosts(id),
    device_ids JSONB DEFAULT '[]',            -- 关联的设备ID列表
    
    status VARCHAR(20) DEFAULT 'idle',        -- idle, busy, offline, maintenance, reserved
    current_task_id UUID,                     -- 当前执行的任务
    
    capabilities JSONB DEFAULT '[]',          -- 环境能力
    tags JSONB DEFAULT '[]',
    
    reserved_by UUID REFERENCES users(id),    -- 预约人
    reserved_until TIMESTAMP WITH TIME ZONE,  -- 预约截止时间
    
    project_id UUID REFERENCES projects(id),
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_environments_status ON environments(status);
CREATE INDEX idx_environments_project_id ON environments(project_id);
```

### 3.4 测试用例

```sql
CREATE TABLE test_suites (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    description TEXT,
    
    project_id UUID REFERENCES projects(id),

```

#### test_cases (测试用例表)
```sql
CREATE TABLE test_cases (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    suite_id UUID REFERENCES test_suites(id) ON DELETE CASCADE,
    
    name VARCHAR(500) NOT NULL,
    description TEXT,
    
    alm_case_id VARCHAR(100),                 -- ALM用例ID
    
    priority VARCHAR(20) DEFAULT 'medium',    -- critical, high, medium, low
    case_type VARCHAR(50),                    -- smoke, regression, functional, performance
    
    script_path VARCHAR(500),                 -- 测试脚本相对路径
    class_name VARCHAR(200),                  -- 测试类名
    method_name VARCHAR(200),                 -- 测试方法名
    
    parameters JSONB DEFAULT '{}',            -- 参数化配置
    
    timeout INTEGER DEFAULT 300,              -- 超时时间(秒)
    retry_count INTEGER DEFAULT 0,            -- 重试次数
    
    preconditions TEXT,                       -- 前置条件
    steps TEXT,                               -- 测试步骤
    expected_results TEXT,                    -- 预期结果
    
    tags JSONB DEFAULT '[]',
    capabilities_required JSONB DEFAULT '[]', -- 所需环境能力
    
    is_automated BOOLEAN DEFAULT TRUE,
    status VARCHAR(20) DEFAULT 'active',      -- active, deprecated
    
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_test_cases_suite_id ON test_cases(suite_id);
CREATE INDEX idx_test_cases_alm_case_id ON test_cases(alm_case_id);
CREATE INDEX idx_test_cases_priority ON test_cases(priority);
```

### 3.5 测试任务

#### test_tasks (测试任务表)
```sql
CREATE TABLE test_tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(200) NOT NULL,
    description TEXT,
    
    project_id UUID REFERENCES projects(id),
    
    -- 任务来源
    source VARCHAR(20) DEFAULT 'manual',      -- manual, alm, ci, api
    external_id VARCHAR(100),                 -- 外部系统ID
    
    -- 触发配置
    trigger_type VARCHAR(20) DEFAULT 'immediate', -- immediate, scheduled, cron
    schedule_time TIMESTAMP WITH TIME ZONE,   -- 定时执行时间
    cron_expression VARCHAR(100),             -- CRON表达式
    
    -- 优先级
    priority INTEGER DEFAULT 5,               -- 1-10, 1最高
    
    -- 升级配置
    need_upgrade BOOLEAN DEFAULT FALSE,
    version_strategy VARCHAR(20),             -- latest, specific
    target_version VARCHAR(50),               -- 目标版本
    artifact_id UUID,                         -- 制品ID
    
    -- 环境配置
    env_strategy VARCHAR(20) DEFAULT 'auto',  -- auto, specific
    environment_id UUID REFERENCES environments(id),
    env_requirements JSONB DEFAULT '{}',      -- 环境要求
    
    -- 测试配置
    suite_id UUID REFERENCES test_suites(id),
    case_ids JSONB,                           -- 指定的用例ID列表(null表示全部)
    case_filter JSONB,                        -- 用例筛选条件
    
    -- 执行配置
    parallel_count INTEGER DEFAULT 1,         -- 并行执行数
    timeout INTEGER DEFAULT 3600,             -- 任务超时时间(秒)
    retry_on_failure BOOLEAN DEFAULT FALSE,
    max_retries INTEGER DEFAULT 1,
    
    -- 通知配置
    notify_config JSONB DEFAULT '{}',         -- 通知配置
    
    -- 状态
    status VARCHAR(20) DEFAULT 'pending',     -- pending, queued, preparing, running, 
                                              -- success, failed, cancelled, timeout
    
    -- 时间戳
    queued_at TIMESTAMP WITH TIME ZONE,
    started_at TIMESTAMP WITH TIME ZONE,
    finished_at TIMESTAMP WITH TIME ZONE,
    
    -- 创建信息
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_test_tasks_project_id ON test_tasks(project_id);
CREATE INDEX idx_test_tasks_status ON test_tasks(status);
CREATE INDEX idx_test_tasks_trigger_type ON test_tasks(trigger_type);
CREATE INDEX idx_test_tasks_schedule_time ON test_tasks(schedule_time);
CREATE INDEX idx_test_tasks_created_at ON test_tasks(created_at DESC);
```

#### task_executions (任务执行记录表)
```sql
CREATE TABLE task_executions (

    -- 版本信息
    software_version VARCHAR(50),             -- 执行时的软件版本
    upgraded_from VARCHAR(50),                -- 升级前版本
    
    -- Git信息
    git_repo VARCHAR(500),
    git_branch VARCHAR(100),
    git_commit VARCHAR(100),
    
    -- 执行状态
    status VARCHAR(20) DEFAULT 'pending',     -- pending, preparing, upgrading, 
                                              -- pulling, running, success, failed, cancelled
    
    -- 时间戳
    started_at TIMESTAMP WITH TIME ZONE,
    finished_at TIMESTAMP WITH TIME ZONE,
    duration INTEGER,                         -- 执行时长(秒)
    
    -- 结果统计
    total_cases INTEGER DEFAULT 0,
    passed_cases INTEGER DEFAULT 0,
    failed_cases INTEGER DEFAULT 0,
    skipped_cases INTEGER DEFAULT 0,
    error_cases INTEGER DEFAULT 0,
    
    -- 日志和错误
    log_path VARCHAR(500),                    -- 日志文件路径
    error_message TEXT,
    
    -- 执行上下文
    context JSONB DEFAULT '{}',               -- 执行上下文信息
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_task_executions_task_id ON task_executions(task_id);
CREATE INDEX idx_task_executions_status ON task_executions(status);
CREATE INDEX idx_task_executions_started_at ON task_executions(started_at DESC);
```

### 3.9 定时任务

#### scheduled_jobs (定时任务表)
```sql
CREATE TABLE scheduled_jobs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    
    task_id UUID REFERENCES test_tasks(id) ON DELETE CASCADE,
    
    job_type VARCHAR(20) NOT NULL,            -- once, cron
    
    schedule_time TIMESTAMP WITH TIME ZONE,   -- 单次执行时间
    cron_expression VARCHAR(100),             -- CRON表达式
    timezone VARCHAR(50) DEFAULT 'Asia/Shanghai',
    
    next_run_time TIMESTAMP WITH TIME ZONE,
    last_run_time TIMESTAMP WITH TIME ZONE,
    
    is_enabled BOOLEAN DEFAULT TRUE,
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX idx_scheduled_jobs_next_run_time ON scheduled_jobs(next_run_time);
CREATE INDEX idx_scheduled_jobs_is_enabled ON scheduled_jobs(is_enabled);
```

## 4. 索引策略

### 4.1 常用查询索引
```sql
-- 任务列表查询优化
CREATE INDEX idx_test_tasks_list ON test_tasks(project_id, status, created_at DESC);

-- 执行历史查询优化
CREATE INDEX idx_task_executions_history ON task_executions(task_id, started_at DESC);

-- 用例结果查询优化
CREATE INDEX idx_test_case_results_query ON test_case_results(execution_id, status);

-- 环境状态查询优化
CREATE INDEX idx_environments_available ON environments(project_id, status) WHERE status = 'idle';
```

### 4.2 全文搜索索引

-- 任务名称搜索
CREATE INDEX idx_test_tasks_name_search ON test_tasks USING gin(to_tsvector('simple', name));

CREATE INDEX idx_test_cases_name_search ON test_cases USING gin(to_tsvector('simple', name));
